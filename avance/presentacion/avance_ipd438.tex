% 2. OBJETIVO
\begin{frame}{Objetivo}
    \begin{itemize}
        \item \textbf{Objetivo:} que un espectador que pierde paquetes recupere la imagen en el próximo I-header\footnote{I-header: la misma imagen codificada además como intra, dentro del stream (AGiLE [1]).}, y no en el próximo intra.
        \item \textbf{Ejemplo:} con GoP\footnote{GoP: grupo de frames entre dos intra.} de 90 a 30 fps, el daño dura hasta 3 s. Con un I-header cada 5 P, $\sim$0,2 s.
        \item \textbf{Sin lazo al codificador:} decide el edge\footnote{Edge: servidor de distribución más cercano al espectador.}; el codificador no recibe pedidos.
        \item \textbf{Sin costo con red sana:} en la última milla, el ancho de banda de un GoP largo.
    \end{itemize}
\end{frame}

        \item \textbf{Edge:} guarda el último I-header y se lo entrega a quien perdió paquetes.
        \item \textbf{Última milla: SRT}\footnote{SRT: Secure Reliable Transport, protocolo abierto sobre UDP para video en vivo [9].} retransmite dentro de un plazo fijo; lo que no alcanza se descarta, y el edge sabe a quién le faltó.
    \end{itemize}
\end{frame}

% 4. CODIFICACIÓN H.264
\begin{frame}{Codificación H.264: I, P e IDR}
    \begin{itemize}
        \item \textbf{Intra (I):} imagen completa. Se decodifica sola; es grande.
        \item \textbf{Predicho (P):} solo lo que cambió respecto del frame anterior. Es pequeño, pero depende del anterior.
        \item \textbf{IDR:} un intra que además reinicia el decodificador.
        \item \textbf{Perder un intra es grave:} cada P se arma sobre el anterior, así que el error se arrastra hasta el próximo intra. Con GoP 90, hasta 3 s.
    \end{itemize}
\end{frame}

    \end{tikzpicture}

    \vspace{0.5cm}
    \raggedright
    \begin{itemize}
        \item \textbf{Una NAL\footnote{NAL: Network Abstraction Layer, la unidad que H.264 entrega a la red.} por slice}, no por macrobloque.
        \item \textbf{PES\footnote{PES: Packetized Elementary Stream, del estándar MPEG-TS.}:} le pone la hora a cada frame; después se corta en paquetes TS de 188 B.
        \item \textbf{También son NAL:} SPS/PPS (configuración del video) y SEI\footnote{SEI: Supplemental Enhancement Information. Datos extra que el decodificador puede ignorar.} (datos extra). Ahí viaja el I-header.
    \end{itemize}
\end{frame}

% 6. ESTADO DEL ARTE: AGILE
\begin{frame}{Estado del Arte: AGiLE [1]}
    \begin{itemize}
        \item \textbf{Problema que resuelve:} con GoP largo, un espectador nuevo espera el próximo intra para ver algo.
        \item \textbf{Idea:} un I-header en cada frontera de pseudo-GOP (cada 2 s en el paper), dentro de una SEI del P.
        \item \textbf{Edge:} se lo entrega solo a espectadores nuevos; al conectado se lo descarta.
        \item \textbf{Indicadores:} BD-rate\footnote{BD-rate: diferencia de bits a igual calidad.} $-3{,}5$\,\%; datos de ingest $+2{,}8$\,\%; calidad al entrar por un I-header $-1{,}1$ VMAF.
        \item \textbf{¿Esperamos mejorarlos?} No: reutilizamos sus I-headers para recuperar pérdidas.
    \end{itemize}
\end{frame}

% 7. ESTADO DEL ARTE: KEYFRAME INJECTION
\begin{frame}{Estado del Arte: Keyframe Injection [2, 3]}
    \begin{itemize}
        \item \textbf{H. Mareen, G. Van Wallendael et al.} Universidad de Gante, 2021--2024.
        \item \textbf{Qué hacen:} un flujo aparte, solo de keyframes. Quien entra o pierde paquetes pide uno y sigue con el stream normal.
        \item \textbf{Indicadores:} caída de calidad $<0{,}94$ VMAF (HEVC). Cada keyframe genera un peak de bitrate; en 2023 lo reducen inyectando P en vez de I [4].
        \item \textbf{Postura:} nos respalda, porque ya probaron que un keyframe extra repara la pérdida.
        \item \textbf{Diferencia:} sin flujo aparte (el I-header va en SEI) y sin pedido del reproductor (decide el edge).
    \end{itemize}
\end{frame}

% 8. PROBLEMA
\begin{frame}{Problema}
    \begin{itemize}
        \item \textbf{Hoy, para cortar el daño:} pedir un intra al codificador (lazo, PLI\footnote{PLI: Picture Loss Indication, el receptor pide un intra.}) o mandar intra siempre (GoP corto, se paga aunque la red esté sana).
        \item \textbf{FrameTrace [5]:} con 1\,\% de pérdida, GoP 0,5 s da 72,6 VMAF; GoP 2 s, 55,7. Refrescar seguido salva la calidad, pero cuesta.
        \item \textbf{Anomalía:} AGiLE ya deja I-headers en el edge\ldots\ y los descarta justo para quien perdió paquetes.
    \end{itemize}

    \vspace{0.6cm}
    {\itshape \small \textcolor{SentricaBlue}{
    \textbf{Idea fuerte:} el mecanismo para recuperarse de la pérdida ya está en la red, ya está pagado, y hoy se descarta.
    }\par}
\end{frame}

% 9. CÓMO LO MEDIMOS
\begin{frame}{Cómo lo Medimos}
    \begin{itemize}
        \item \textbf{Duración:} frames desde la pérdida hasta la imagen limpia.
        \item \textbf{Calidad, VMAF\footnote{VMAF: Video Multimethod Assessment Fusion (Netflix, 2016).}:} compara cada frame recibido con el original y predice cuánto lo notaría una persona, de 0 a 100.
        \item \textbf{Congelamiento, FRR\footnote{FRR: Frame Render Rate, frames mostrados / frames enviados [5].}:} congelar no cuenta como recuperar.
        \item \textbf{Costo:} bits extra en el ingest y en la última milla.
        \item \textbf{Referencia limpia:} un video de prueba sin comprimir (y4m) entra al codificador y sirve de original.
    \end{itemize}
\end{frame}
