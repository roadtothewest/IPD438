\begin{frame}{Codificación H.264: I, P e IDR}
    \begin{itemize}
        \item \textbf{Intra (I):} imagen completa. Se decodifica sola; es grande.

% 5. CÓMO SE RECONSTRUYE
\begin{frame}{Cómo se Reconstruye el Video}
    \begin{itemize}
        \item \textbf{El decodificador:} lee la configuración (SPS/PPS) y decodifica el intra solo.
        \item \textbf{Cada P:} por cada macrobloque, copia un bloque del frame anterior (donde indica el vector de movimiento) y le suma una pequeña corrección.
        \item \textbf{Lo que se pierde es un paquete SRT} (7 TS): el slice se daña desde ese punto hasta su final; los macrobloques anteriores se salvan.
        \item \textbf{Con un slice por fila de macrobloques}, sabemos en qué frame (tiempo) y en qué fila (espacio) está el daño.
    \end{itemize}
\end{frame}

\begin{frame}{Problema y Motivación}
    \begin{itemize}
        \item \textbf{Problema:} una pérdida se arrastra hasta el próximo intra.
        \item \textbf{Motivación:} sin propagación del error no hay anomalía; el decodificador solo la disimula.
        \item \textbf{¿GoP corto?} El daño dura menos, pero todos los espectadores pagan más bits.
    \end{itemize}

% 8. ESTADO DEL ARTE: AGILE
\begin{frame}{Estado del Arte: AGiLE [1]}
    {\scriptsize\color{gray} C. Chen, W. Yin, Z. Huang, S. Shi (ByteDance). ACM MMSys 2024.\par}
    \vspace{0.1cm}
    \begin{itemize}
        \item \textbf{Resuelve:} con GoP largo, quien entra espera el próximo intra.
        \item \textbf{Idea:} un I-header cada 2 s, dentro de una SEI del P.
        \item \textbf{Edge:} se lo da solo a quien entra; al resto se lo descarta.
        \item \textbf{Indicadores:} BD-rate\footnote{BD-rate: Bjøntegaard Delta rate, la diferencia promedio de bits a igual calidad.} $-3{,}5$\,\%; subida $+2{,}8$\,\%; $-1{,}1$ VMAF\footnote{VMAF: Video Multimethod Assessment Fusion, calidad percibida de 0 a 100.} al entrar por un I-header.
        \item \textbf{¿Esperamos mejorarlos?} No: reutilizamos sus I-headers para recuperar pérdidas.
    \end{itemize}

    \vspace{0.2cm}
    {\itshape \small \textcolor{SentricaBlue}{
    \textbf{Idea fuerte:} el mecanismo para recuperarse de la pérdida ya está en la red, ya está pagado, y hoy se descarta.
    }\par}
\end{frame}

% 9. ESTADO DEL ARTE: KEYFRAME INJECTION
\begin{frame}{Estado del Arte: Keyframe Injection [2, 3]}
    {\scriptsize\color{gray} H. Mareen, G. Van Wallendael et al. (Universidad de Gante), 2021--2024.\par}
    \vspace{0.1cm}
    \begin{itemize}
        \item \textbf{Qué hacen:} un flujo aparte, solo de keyframes. Quien entra o pierde paquetes pide uno y sigue con el stream normal.
        \item \textbf{Indicadores:} caída de calidad $<0{,}94$ VMAF; cada keyframe inyectado genera un peak de bitrate.
        \item \textbf{2023:} achican ese peak reparando con un P que se apoya en un frame anterior a la pérdida, en vez de un I [4].
        \item \textbf{Postura:} nos respalda, porque ya probaron que un keyframe extra repara la pérdida.
        \item \textbf{Diferencia:} sin flujo aparte (el I-header va en SEI) y sin pedido del reproductor (decide el edge).
    \end{itemize}
\end{frame}

        \item \textbf{Resultado:} con un I-header cada 5 P, el daño dura como máximo 5 P; el intra extra solo le llega a quien perdió.
    \end{itemize}

% 11. OBJETIVO
\begin{frame}{Objetivo}
    \begin{itemize}
        \item \textbf{Objetivo:} que un espectador que pierde paquetes recupere la imagen en el próximo I-header, y no en el próximo intra.
        \item \textbf{Ejemplo:} con GoP 90 a 30 fps, el daño dura hasta 3 s. Con un I-header cada 5 P, $\sim$0,2 s.
        \item \textbf{Sin pedidos al codificador por cada pérdida:} decide el edge.
        \item \textbf{Sin costo con red sana:} el espectador que no pierde recibe lo mismo que con GoP 90.
    \end{itemize}
\end{frame}

% 13. CÓMO LO MEDIMOS
\begin{frame}{Cómo lo Medimos}
    \begin{itemize}
        \item \textbf{Referencia:} el mismo stream decodificado sin pérdidas (integridad original).
        \item \textbf{Integridad temporal:} las diferencias entre frames consecutivos deben parecerse a las de la referencia.
        \item \textbf{Integridad de imagen:} PSNR\footnote{PSNR: Peak Signal-to-Noise Ratio. Error píxel a píxel contra la referencia, en dB.} de cada frame contra la referencia.
        \item \textbf{Duración y congelamiento:} frames hasta recuperar la integridad; FRR\footnote{FRR: Frame Render Rate, frames mostrados / frames enviados [5].}.
        \item \textbf{Costo:} bits extra hacia el espectador y en la subida (BD-rate).
    \end{itemize}
\end{frame}

% 14. RESULTADOS A PRODUCIR
\begin{frame}{Resultados a Producir}
    \textbf{Reproducir}
    \begin{itemize}
        \item La degradación al entrar por un I-header que reporta AGiLE, ahora en H.264.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Producir: misma pérdida en SRT}
    \begin{itemize}
        \item \textbf{Comparar:} GoP 90 \enspace / \enspace GoP 30 \enspace / \enspace GoP 90 + I-header cada 5 P.
        \item \textbf{Variable a barrer:} cada cuántos P va un I-header.
    \end{itemize}

    \vspace{0.6cm}
    {\itshape \small \textcolor{SentricaBlue}{
    \textbf{Impacto esperado:} quien pierde paquetes se recupera como con un GoP corto, y solo él recibe bits extra; quien no pierde recibe lo mismo que con GoP 90.
    }\par}
