\section{Contenidos previos}

\begin{frame}{Cómo se comprime y se transmite el video}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.2em}
        \item Video = imágenes (\textbf{frames}), 30 por segundo.
        \item 1080p sin comprimir: $\sim$750 Mb/s. Con el códec \textbf{H.264}: $\sim$5 Mb/s.
        \item Truco: frames vecinos se parecen.
        \begin{itemize}
            \item \textbf{I} (intra): imagen completa. Grande, se decodifica sola.
            \item \textbf{P} (predicho): solo lo que cambió. Pequeño, depende del anterior.
            \item \textbf{IDR}: un I que además reinicia el decodificador.
        \end{itemize}
        \item \textbf{GoP} (\emph{Group of Pictures}): de un I al siguiente.
        \item Viaja en paquetes de $\sim$1300 B: un I ocupa decenas; un P, pocos.
    \end{itemize}

\begin{frame}{¿Por qué perder un intra es tan grave?}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.1em}
        \item Cada P se arma sobre el anterior: el error se arrastra hasta el próximo I (\textbf{propagación}).
        \item Se pierde un P $\rightarrow$ daño hasta el fin del GoP.
        \item Se pierde el I $\rightarrow$ \textbf{todo el GoP} (1 s con GoP 30 a 30 fps).
        \item Se pierde el \textbf{IDR} $\rightarrow$ \textbf{peor caso}: además se pierde el reinicio; la imagen puede congelarse.
        \item El I es el más grande $\rightarrow$ el más expuesto a perder un paquete.
        \item \textbf{Ocultar} (congelar, copiar el frame anterior) disimula el daño, no lo elimina.
    \end{itemize}

\section{Estado del arte}

\begin{frame}{AGiLE [2]: GoP largo con puntos de entrada cada 2 s}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.2em}
        \item \textbf{Resuelve}: GoP largo ahorra bits, pero quien se conecta espera el próximo I.
        \item \textbf{Idea}: GoP de codificación largo (10 s), \textbf{pseudo-GOP} de transmisión corto (2 s).
        \item \textbf{I-header}: en cada frontera, la misma imagen codificada además como I.
        \item Viaja \textbf{dentro de una SEI} del P: un decodificador normal la ignora.
    \end{itemize}

    \vspace{0.3em}

        \draw[-Latex, thick, AlertRed, dashed] (edge.east) -- (con.west);

    \vspace{0.5em}
    \raggedright
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.25em}
        \item \textbf{Edge}: servidor de la red de distribución (\textbf{CDN}) más cercano al espectador.
        \item Lee la SEI, guarda el I-header y lo entrega \textbf{solo a quien recién se conecta}.
        \item Ahorra $\sim$3,5 \% de bits a igual calidad; suma +2,8 \% de datos hacia el CDN.
        \item Quien entra por un I-header ve una diferencia leve: $\sim$1 punto de \textbf{VMAF} (calidad percibida, 0--100).
    \end{itemize}

    \cita{[2] C. Chen, W. Yin, Z. Huang, S. Shi. \emph{AGiLE: Enhancing Adaptive GOP in Live Video Streaming}. ACM MMSys '24, pp. 34--44.}
\end{frame}

\begin{frame}{FrameTrace [1] y cómo se recupera uno hoy}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.25em}
        \item \textbf{FrameTrace}: con 1 \% de pérdida, GoP 0,5 s $\rightarrow$ VMAF 72,6; GoP 2 s $\rightarrow$ 55,7.
        \item Acorta el GoP si hay pérdida, pero el receptor debe avisarle al codificador (\textbf{lazo}).
    \end{itemize}

\section{SRT}

\begin{frame}{Transporte: SRT}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.25em}
        \item \textbf{SRT} (\emph{Secure Reliable Transport}): protocolo abierto sobre UDP para video en vivo [9].
        \item Latencia fija (120 ms por defecto): en ese plazo, el receptor pide lo que falta (NAK) y el emisor lo reenvía.
        \item Lo que no llega a tiempo \textbf{se descarta} $\rightarrow$ esa pérdida llega al decodificador.
        \item Lleva MPEG-TS (paquetes de 188 B) y cifra con AES.
        \item \textbf{Ventaja}: el emisor (nuestro edge) sabe qué descartó $\rightarrow$ sabe a quién darle el I-header, \textbf{sin preguntarle al codificador}.
    \end{itemize}

    \vspace{0.4em}

\section{Problema}

\begin{frame}{Problema: el punto de quiebre}
    \small
    \begin{itemize}

\section{Objetivo}

\begin{frame}{Objetivo}
    \small
    \begin{itemize}
        \setlength{\itemsep}{0.2em}
        \item Demostrar que entregar el I-header a quien perdió paquetes corta el daño en la próxima frontera ($\leq$ 2 s) y no en el próximo I ($\leq$ 10 s), \textbf{sin lazo al codificador}.
        \item Meta: ancho de banda de GoP largo, daño cercano al de GoP corto.
        \item Métricas: \textbf{duración} (frames dañados) $\cdot$ \textbf{severidad} (VMAF) $\cdot$ \textbf{congelamiento} (FRR: frames mostrados / enviados).

\section{Costos}

\begin{frame}{Costos: qué perdemos}
    \small
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{0.35em}
                \item \textbf{Deriva}: entrar por un I-header deja un error leve hasta el próximo I.
                \item \textbf{Ráfaga}: un I pesa mucho más que un P y se envía cuando el enlace ya pierde (por eso LTR evita intras).
                \item \textbf{Doble codificación} en el STM32: cómputo, memoria y energía.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{0.35em}
                \item \textbf{Ingest}: +2,8 \% de datos hacia el CDN (AGiLE, H.265); en H.264, por medir.
                \item \textbf{Estado en el edge}: sin lazo al codificador $\neq$ sin estado.
                \item \textbf{Granularidad}: solo se recupera en fronteras de pseudo-GOP.
                \item \textbf{Contenido}: con mucho movimiento, el GoP largo rinde menos.
                \item \textbf{Compatibilidad}: decodificadores que no arrancan desde un I no-IDR.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
