\newcommand{\secuencia}[5]{%
  \node[anchor=east, font=\scriptsize] at (-0.2, #1+0.17) {#5};
  \pgfmathtruncatemacro{\proxI}{#4 < 0 ? -1 : (floor(#4/#2)+1)*#2}%
  \pgfmathtruncatemacro{\proxH}{#4 < 0 ? -1 : (#3 == 1 ? min((floor(#4/6)+1)*6, \proxI) : \proxI)}%
  \foreach \i in {0,...,90}{%
    \pgfmathtruncatemacro{\esI}{mod(\i,#2) == 0 ? 1 : 0}%
    \pgfmathtruncatemacro{\estado}{\esI == 1 ? 1 : ((#4 >= 0 && \i >= #4 && \i < \proxH) ? 2 : ((#4 >= 0 && \i >= \proxH && \i < \proxI) ? 3 : 0))}%
    \ifcase\estado \def\col{Suave}\or \def\col{SentricaBlue}\or \def\col{Rojo}\or \def\col{Rojo!25}\fi
    \fill[\col] (\i*0.115, #1) rectangle ++(0.1, 0.34);
  }%
  \ifnum#3=1
    \foreach \k in {6,12,...,90}{%
      \pgfmathtruncatemacro{\esI}{mod(\k,#2) == 0 ? 1 : 0}%
      \ifnum\esI=0 \fill[SentricaBlue] (\k*0.115+0.05, #1+0.42) circle (0.04);\fi
    }%
  \fi
  \ifnum#4>-1 \node[font=\tiny] at (#4*0.115+0.05, #1+0.5) {$\blacktriangledown$};\fi
}

  \foreach \s/\f in {0/0, 1/30, 2/60, 3/90}{\node[font=\tiny, text=gray] at (\f*0.115+0.05, #1) {\s\,s};}
}

% 2. CODIFICACIÓN
\begin{frame}{Codificación H.264: I, P e IDR}
    \begin{itemize}
        \item \textbf{Intra (I):} imagen completa. Se decodifica sola; es grande.
        \item \textbf{Predicho (P):} solo lo que cambió respecto del frame anterior. Es pequeño, pero depende del anterior.
        \item \textbf{IDR:} un intra que además reinicia el decodificador.
        \item \textbf{GoP\footnote{GoP: Group of Pictures, los frames entre dos intra.}:} cada cuántos frames va un intra.
    \end{itemize}

    \vspace{0.3cm}
    \centering
    \begin{tikzpicture}

    {\scriptsize 30 fps: GoP 30 = un intra por segundo; GoP 90 = uno cada 3 s.}
\end{frame}

    \vspace{0.4cm}
    \raggedright
    \begin{itemize}
        \item \textbf{Una NAL\footnote{NAL: Network Abstraction Layer, la unidad que H.264 entrega a la red.} por slice}, no por macrobloque.
        \item \textbf{PES\footnote{PES: Packetized Elementary Stream. Su marca de tiempo (PTS) cuenta ticks de 90 kHz desde el inicio del stream; no es hora del reloj.}:} encabezado con la marca de tiempo del frame + todas sus NAL.
        \item \textbf{TS\footnote{TS: Transport Stream, el contenedor de MPEG.}:} corta el PES en paquetes de 188 B.
        \item \textbf{SRT va dentro de UDP:} 7 TS por paquete, cada uno con número de secuencia y marca de tiempo.
    \end{itemize}
\end{frame}

% 4. CÓMO SE RECONSTRUYE
\begin{frame}{Cómo se Reconstruye el Video}
    \begin{itemize}
        \item \textbf{El decodificador:} lee la configuración (SPS/PPS) y decodifica el intra solo.
        \item \textbf{Cada P:} por cada macrobloque, copia un bloque del frame anterior (donde indica el vector de movimiento) y le suma una pequeña corrección.
        \item \textbf{Lo muestra en su momento:} según la marca de tiempo del PES.
        \item \textbf{Lo que se pierde es un paquete SRT} (7 TS): el slice se daña desde ese punto hasta su final; los macrobloques anteriores se salvan.
        \item \textbf{Con un slice por fila de macrobloques}, sabemos en qué frame (tiempo) y en qué fila (espacio) está el daño.
    \end{itemize}
\end{frame}

% 6. ESTADO DEL ARTE: AGILE
\begin{frame}{Estado del Arte: AGiLE [1]}
    \begin{itemize}
        \item \textbf{Resuelve:} con GoP largo, un espectador nuevo espera el próximo intra.
        \item \textbf{Idea:} un I-header cada 2 s, dentro de una SEI del P.
        \item \textbf{Edge:} lo entrega solo a espectadores nuevos; al conectado se lo descarta.
        \item \textbf{Indicadores:} BD-rate\footnote{BD-rate: diferencia de bits a igual calidad.} $-3{,}5$\,\%; subida $+2{,}8$\,\%; $-1{,}1$ VMAF\footnote{VMAF: calidad percibida de 0 a 100 (Netflix). Es la métrica que usan estos papers.} al entrar por un I-header.
        \item \textbf{¿Esperamos mejorarlos?} No: reutilizamos sus I-headers para recuperar pérdidas.
    \end{itemize}

    \vspace{0.2cm}
    {\itshape \small \textcolor{SentricaBlue}{
    \textbf{Idea fuerte:} el mecanismo para recuperarse de la pérdida ya está en la red, ya está pagado, y hoy se descarta.
    }\par}
\end{frame}

% 7. ESTADO DEL ARTE: KEYFRAME INJECTION
\begin{frame}{Estado del Arte: Keyframe Injection [2, 3]}
    \begin{itemize}
        \item \textbf{H. Mareen, G. Van Wallendael et al.} Universidad de Gante, 2021--2024.
        \item \textbf{Qué hacen:} un flujo aparte, solo de keyframes. Quien entra o pierde paquetes pide uno y sigue con el stream normal.
        \item \textbf{Indicadores:} caída de calidad $<0{,}94$ VMAF. Cada keyframe genera un peak de bitrate; en 2023 lo reducen inyectando P en vez de I [4].
        \item \textbf{Postura:} nos respalda, porque ya probaron que un keyframe extra repara la pérdida.
        \item \textbf{Diferencia:} sin flujo aparte (el I-header va en SEI) y sin pedido del reproductor (decide el edge).
    \end{itemize}
\end{frame}

% 8. PROBLEMA, MOTIVACIÓN Y PUNTO DE QUIEBRE
\begin{frame}{Problema, Motivación y Punto de Quiebre}
    \begin{itemize}
        \item \textbf{Problema:} una pérdida se arrastra hasta el próximo intra.
        \item \textbf{Motivación:} sin propagación del error no hay anomalía; el decodificador solo la disimula.
        \item \textbf{Punto de quiebre:} con un I-header cada 5 P, el daño dura como máximo 5 P, sin pedirle nada al codificador.
    \end{itemize}

% 9. OBJETIVO
\begin{frame}{Objetivo}
    \begin{itemize}
        \item \textbf{Objetivo:} que un espectador que pierde paquetes recupere la imagen en el próximo I-header, y no en el próximo intra.
        \item \textbf{Ejemplo:} con GoP 90 a 30 fps, el daño dura hasta 3 s. Con un I-header cada 5 P, $\sim$0,2 s.
        \item \textbf{Sin lazo al codificador:} decide el edge; el codificador no recibe pedidos.
        \item \textbf{Sin costo con red sana:} hacia el espectador, el ancho de banda de un GoP largo.
    \end{itemize}
\end{frame}

    \vspace{0.1cm}
    \raggedright
    \begin{itemize}
        \item \textbf{Modo A, siempre:} I-header cada 5 P; el edge lo reenvía solo a quien perdió.
        \item \textbf{Modo B, a pedido:} el edge detecta pérdida y activa el flujo de I-headers\footnote{Ahorra subida con red sana, pero agrega un aviso del edge al codificador.}.
        \item \textbf{SRT\footnote{SRT: Secure Reliable Transport, protocolo abierto sobre UDP para video en vivo [9].}:} secuencia y marca de tiempo en cada paquete $\rightarrow$ sabemos en qué frame y en qué slice se perdió.
    \end{itemize}
\end{frame}

% 11. CÓMO LO MEDIMOS
\begin{frame}{Cómo lo Medimos}
    \begin{itemize}
        \item \textbf{Referencia:} el mismo stream decodificado sin pérdidas (integridad original).
        \item \textbf{Integridad temporal:} las diferencias entre frames consecutivos deben parecerse a las de la referencia\footnote{Misma base que la información temporal (TI) de ITU-T P.910 [11] y que la parte temporal de ST-RRED [10].}.
        \item \textbf{Integridad de imagen:} PSNR\footnote{PSNR: relación señal a ruido de pico, en dB.} de cada frame contra la referencia.
        \item \textbf{Duración y congelamiento:} frames hasta recuperar la integridad; FRR\footnote{FRR: Frame Render Rate, frames mostrados / frames enviados [5].}.
        \item \textbf{Costo:} bits extra hacia el espectador y en la subida (BD-rate).
    \end{itemize}
\end{frame}
