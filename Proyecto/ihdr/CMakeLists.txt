cmake_minimum_required(VERSION 3.16)
project(ihdr CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
add_executable(ihdr_sim host/main.cpp)
add_executable(test_core host/test_core.cpp)
# El núcleo debe compilar sin excepciones ni RTTI (como en el STM32).
set_source_files_properties(host/test_core.cpp PROPERTIES COMPILE_OPTIONS "-fno-exceptions;-fno-rtti")
